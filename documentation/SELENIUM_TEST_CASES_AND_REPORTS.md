import pytest
import time
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

class TestLogin:
    def setup_method(self, method):
        self.driver = webdriver.Chrome()
        self.driver.maximize_window()
        self.wait = WebDriverWait(self.driver, 10)

    def teardown_method(self, method):
        self.driver.quit()

    def test_login(self):
        # Navigate to NEUROSPECTRA login portal
        self.driver.get("http://localhost:3000/login")
        
        # Wait for email field and enter registered clinician/admin credentials
        email_field = self.wait.until(
            EC.visibility_of_element_located((By.ID, "login-email"))
        )
        email_field.clear()
        email_field.send_keys("admin@neurospectra.org")
        
        # Wait for password field and enter password
        password_field = self.wait.until(
            EC.visibility_of_element_located((By.ID, "login-password"))
        )
        password_field.clear()
        password_field.send_keys("admin123")
        
        # Click Sign In button
        login_btn = self.wait.until(
            EC.element_to_be_clickable((By.ID, "btn-login-submit"))
        )
        login_btn.click()
        
        # Verify successful login and loading of authenticated role dashboard
        self.wait.until(
            EC.presence_of_element_located((By.CLASS_NAME, "app-sidebar"))
        )
        time.sleep(1)
        body_text = self.driver.find_element(By.TAG_NAME, "body").text
        assert "Admin Overview" in body_text or "TOTAL CHILDREN" in body_text

import pytest
import time
from datetime import datetime, timedelta
from selenium import webdriver
from selenium.webdriver.common.by import By

from selenium.webdriver.support import expected_conditions as EC

class TestAppointmentScheduling:
    def setup_method(self, method):
        self.driver = webdriver.Chrome()
        self.driver.maximize_window()
        self.wait = WebDriverWait(self.driver, 10)

    def teardown_method(self, method):
        self.driver.quit()

    def test_appointment_booking_and_slot_hiding(self):
        # Authenticate as Receptionist (Spot Booking for all clinic children)
        self.driver.get("http://localhost:3000/login")
        self.driver.execute_script("window.neuroAuth.switchDemoRole('Receptionist');")
        self.driver.execute_script("window.navigateTo('appointments');")
        time.sleep(0.8)
        
        # Open Booking Modal
        self.driver.execute_script("window.showBookAppointmentModal();")
        time.sleep(0.5)
        
        # Verify universal child access
        child_select = self.driver.find_element(By.ID, "apt-child")
        options = [o.text for o in child_select.find_elements(By.TAG_NAME, "option")]
        assert len(options) >= 3, "Receptionist must have spot booking access for all children"
        
        # Select tomorrow's date
        tomorrow = (datetime.now() + timedelta(days=1)).strftime("%Y-%m-%d")
        self.driver.execute_script(f"""
            const d = document.getElementById('apt-date');
            if (d) {{ d.value = '{tomorrow}'; window.updateAvailableAppointmentSlots(); }}
        """)
        time.sleep(0.5)
        
        # Select time slot and confirm booking
        time_select = self.driver.find_element(By.ID, "apt-time")
        available_slots = [o.text for o in time_select.find_elements(By.TAG_NAME, "option")]
        assert len(available_slots) > 0, "No available slots"
        
        booked_slot = available_slots[0].split(' - ')[0].strip()
        self.driver.find_element(By.ID, "apt-notes").send_keys("Receptionist Automated Booking")
        self.driver.find_element(By.ID, "btn-confirm-booking").click()
        time.sleep(0.8)
        
        # Reopen modal for same date and verify booked slot is dynamically excluded
        self.driver.execute_script("window.showBookAppointmentModal();")
        time.sleep(0.5)
        self.driver.execute_script(f"""
            const d = document.getElementById('apt-date');
            if (d) {{ d.value = '{tomorrow}'; window.updateAvailableAppointmentSlots(); }}
        """)
        time.sleep(0.5)
        
        slots_after = [o.text for o in self.driver.find_element(By.ID, "apt-time").find_elements(By.TAG_NAME, "option")]
        assert not any(booked_slot in s for s in slots_after), f"Double-booking bug: {booked_slot} still visible"
